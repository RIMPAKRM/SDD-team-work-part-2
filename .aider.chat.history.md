````
import { addDays, format, getDay, startOfDay } from 'date-fns';
import { ru } from 'date-fns/locale';

const DAY_MAP = {
  "Воскресенье": 0,
  "Понедельник": 1,
  "Вторник": 2,
  "Среда": 3,
  "Четверг": 4,
  "Пятница": 5,
  "Суббота": 6
};

/**
 * Генерирует список слотов занятий на ближайшие 14 дней
 * @param {Array} groups - Список групп
 * @returns {Array} - Список объектов { date, dateStr, dayName, ...groupData }
 */
export const getUpcomingSlots = (groups) => {
  const today = startOfDay(new Date());
  const slots = [];

  for (let i = 0; i < 14; i++) {
    const date = addDays(today, i);
    const dayIndex = getDay(date);
    const dateStr = format(date, 'dd.MM', { locale: ru });
    const dayName = format(date, 'EEEE', { locale: ru });

    groups.forEach(group => {
      group.schedule.forEach(s => {
        if (DAY_MAP[s.day] === dayIndex) {
          slots.push({
            ...s,
            date: date,
            dateStr: dateStr,
            dayName: dayName,
            groupId: group.id,
            groupLevel: group.level,
            groupTeacher: group.teacher,
            groupCapacity: group.capacity
          });
        }
      });
    });
  }
  return slots;
};
````

````
import groups from '../../data/groups.json';
import { getUpcomingSlots } from '../../utils/dateUtils';

export default function SchedulePage() {
  const upcomingSlots = getUpcomingSlots(groups);

  // Группируем слоты по датам для отображения
  const slotsByDate = upcomingSlots.reduce((acc, slot) => {
    const dateKey = slot.dateStr;
    if (!acc[dateKey]) {
      acc[dateKey] = {
        dateStr: slot.dateStr,
        dayName: slot.dayName,
        items: []
      };
    }
    acc[dateKey].items.push(slot);
    return acc;
  }, {});

  const sortedDates = Object.values(slotsByDate).sort((a, b) => {
    // Простая сортировка по дате (dd.MM)
    const [d1, m1] = a.dateStr.split('.').map(Number);
    const [d2, m2] = b.dateStr.split('.').map(Number);
    return (m1 * 31 + d1) - (m2 * 31 + d2);
  });

  return (
    <div className="p-4 max-w-6xl mx-auto">
      <h1 className="text-3xl font-bold mb-8 text-gray-900">Расписание занятий на 14 дней</h1>
      
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {sortedDates.length === 0 ? (
          <p className="text-gray-500">Нет доступных занятий на ближайшие 14 дней.</p>
````

````
import { addDays, format, getDay, startOfDay } from 'date-fns';
import { ru } from 'date-fns/locale';

const DAY_MAP = {
  "Воскресенье": 0,
  "Понедельник": 1,
  "Вторник": 2,
  "Среда": 3,
  "Четверг": 4,
  "Пятница": 5,
  "Суббота": 6
};

/**
 * Генерирует список слотов занятий на ближайшие 14 дней
 * @param {Array} groups - Список групп
 * @param {Date} startDate - Начальная дата (по умолчанию сегодня)
 * @returns {Array} - Список объектов { date, dateStr, dayName, ...slotData }
 */
export const getUpcomingSlots = (groups, startDate = new Date()) => {
  const today = startOfDay(startDate);
  const slots = [];

  for (let i = 0; i < 14; i++) {
    const date = addDays(today, i);
    const dayIndex = getDay(date);
    const dateStr = format(date, 'dd.MM', { locale: ru });
    const dayName = format(date, 'EEEE', { locale: ru });

    groups.forEach(group => {
      group.schedule.forEach(s => {
        if (DAY_MAP[s.day] === dayIndex) {
          slots.push({
            ...s,
            date: date,
            dateStr: dateStr,
            dayName: dayName,
            groupId: group.id,
            groupLevel: group.level,
            groupTeacher: group.teacher,
            groupCapacity: group.capacity
          });
        }
      });
    });
  }
  return slots;
};
````

````
import { addDays, format, getDay, startOfDay } from 'date-fns';
import { ru } from 'date-fns/locale';

const DAY_MAP = {
  "Воскресенье": 0,
  "Понедельник": 1,
  "Вторник": 2,
  "Среда": 3,
  "Четверг": 4,
  "Пятница": 5,
  "Суббота": 6
};

/**
 * Генерирует список слотов занятий на ближайшие 14 дней
 * @param {Array} groups - Список групп
 * @param {Date} startDate - Начальная дата (по умолчанию сегодня)
 * @returns {Array} - Список объектов { date, dateStr, dayName, ...slotData }
 */
export const getUpcomingSlots = (groups, startDate = new Date()) => {
  // Создаем копию даты, чтобы не мутировать входящий объект
  const today = startOfDay(new Date(startDate.getTime()));
  const slots = [];

  for (let i = 0; i < 14; i++) {
    const date = addDays(today, i);
    const dayIndex = getDay(date);
    const dateStr = format(date, 'dd.MM', { locale: ru });
    const dayName = format(date, 'EEEE', { locale: ru });

    groups.forEach(group => {
      group.schedule.forEach(s => {
        if (DAY_MAP[s.day] === dayIndex) {
          slots.push({
            ...s,
            date: date,
            dateStr: dateStr,
            dayName: dayName,
            groupId: group.id,
            groupLevel: group.level,
            groupTeacher: group.teacher,
            groupCapacity: group.capacity
          });
        }
      });
    });
  }
  return slots;
};
````

````
import { addDays, format, getDay, startOfDay } from 'date-fns';
import { ru } from 'date-fns/locale';

const DAY_MAP = {
  "Воскресенье": 0,
  "Понедельник": 1,
  "Вторник": 2,
  "Среда": 3,
  "Четверг": 4,
  "Пятница": 5,
  "Суббота": 6
};

/**
 * Генерирует список слотов занятий на ближайшие 14 дней
 * @param {Array} groups - Список групп
 * @param {Date} startDate - Начальная дата (по умолчанию сегодня)
 * @returns {Array} - Список объектов { date, dateStr, dayName, ...slotData }
 */
export const getUpcomingSlots = (groups, startDate = new Date()) => {
  // Создаем копию даты, чтобы не мутировать входящий объект
  const today = startOfDay(new Date(startDate.getTime()));
  const slots = [];

  for (let i = 0; i < 14; i++) {
    const date = addDays(today, i);
    const dayIndex = getDay(date);
    const dateStr = format(date, 'dd.MM', { locale: ru });
    const dayName = format(date, 'EEEE', { locale: ru });

    groups.forEach(group => {
      group.schedule.forEach(s => {
        if (DAY_MAP[s.day] === dayIndex) {
          slots.push({
            ...s,
            date: date,
            dateStr: dateStr,
            dayName: dayName,
            groupId: group.id,
            groupLevel: group.level,
            groupTeacher: group.teacher,
            groupCapacity: group.capacity
          });
        }
      });
    });
  }
  return slots;
};
````

````
  });
````

````
const STORAGE_KEY = 'trial_signups';

export const getSignups = () => {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  } catch (e) {
    return [];
  }
};

export const saveSignup = (signup) => {
  const signups = getSignups();
````
  
````
  // Помечаем старую активную запись этого пользователя как отмененную
  const updatedSignups = signups.map(s => 
    s.phone === signup.phone && s.status === 'active' 
      ? { ...s, status: 'cancelled' } 
      : s
  );

  // Добавляем новую запись
  updatedSignups.push({ 
    ...signup, 
    id: Date.now().toString(), 
    status: 'active' 
  });
  
  localStorage.setItem(STORAGE_KEY, JSON.stringify(updatedSignups));
};
````

````
                  const isFull = occupied >= slot.groupCapacity;

  );
````

### 2. Ответы на вопросы аудита

*   **Проверяется ли свободное место непосредственно перед сохранением?**
    Да, в `TrialSignupForm.jsx` внутри `handleSubmit` выполняется проверка `occupied >= slot.groupCapacity`.
*   **Что происходит со старой записью, если новая группа уже заполнена?**
    Если новая группа заполнена, `handleSubmit` прерывает выполнение до вызова `saveSignup`. Старая запись остается в `localStorage` без изменений.
*   **Сохраняется ли старая запись, если пользователь отказывается от перезаписи?**
    Да. Если пользователь нажимает «Нет» в диалоге перезаписи, состояние `existingSignup` сбрасывается, и никаких действий с `localStorage` не производится.
*   **Учитываются ли только записи со статусом active?**
    Да, во всех местах, где происходит подсчёт занятости (в `SchedulePage.jsx` и `TrialSignupForm.jsx`), используется фильтрация `s.status === 'active'`.
*   **Используется ли локальная дата YYYY-MM-DD без преобразования через toISOString()?**
    В текущем коде используется `slot.date.toISOString().split('T')[0]`. Это стандартный способ получения ISO-даты, но он зависит от UTC. Для локального времени Казахстана (UTC+5) это может привести к смещению даты, если время занятия близко к полуночи UTC.
*   **Защищено ли сохранение от двойного нажатия?**
    В текущей реализации нет явного `disabled` состояния для кнопки отправки, но после успешного сохранения `isSuccess` становится `true`, что меняет UI на экран подтверждения, предотвращая повторную отправку.

````
import { useState } from 'react';
import { getSignups, saveSignup } from '../../utils/storage';

export default function TrialSignupForm({ slot, onClose, onSuccess }) {
  const [formData, setFormData] = useState({
    name: '',
    phone: '',
    email: ''
  });
  const [errors, setErrors] = useState({});
  const [existingSignup, setExistingSignup] = useState(null);
  const [isSuccess, setIsSuccess] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const year = slot.date.getFullYear();
  const month = String(slot.date.getMonth() + 1).padStart(2, '0');
  const day = String(slot.date.getDate()).padStart(2, '0');
  const slotDateISO = `${year}-${month}-${day}`;

  const validate = () => {
    const newErrors = {};
    if (!formData.name.trim()) newErrors.name = 'Имя обязательно';
    
    const phoneRegex = /^\+7 \(\d{3}\) \d{3}-\d{2}-\d{2}$/;
    if (!phoneRegex.test(formData.phone)) {
      newErrors.phone = 'Неверный формат: +7 (XXX) XXX-XX-XX';
    }

    if (formData.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email)) {
      newErrors.email = 'Неверный формат email';
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (isSubmitting || !validate()) return;

    setIsSubmitting(true);

    // Проверка занятости
    const allSignups = getSignups();
    const occupied = allSignups.filter(s => 
      s.status === 'active' && 
      s.groupId === slot.groupId && 
      s.date === slotDateISO
    ).length;

    if (occupied >= slot.groupCapacity) {
      alert('К сожалению, на это время мест больше нет.');
      setIsSubmitting(false);
      return;
    }

    // Проверка на дубликат
    const duplicate = allSignups.find(s => s.status === 'active' && s.phone === formData.phone);
    
    if (duplicate && !existingSignup) {
      setExistingSignup(duplicate);
      setIsSubmitting(false);
      return;
    }

    // Сохранение
    saveSignup({
      ...formData,
      groupId: slot.groupId,
      groupLevel: slot.groupLevel,
      groupTeacher: slot.groupTeacher,
      date: slotDateISO,
      time: slot.time
    });
    setIsSuccess(true);
    setIsSubmitting(false);
  };

  const handlePhoneChange = (e) => {
    let value = e.target.value.replace(/\D/g, '');
    if (value.length === 11 && (value.startsWith('7') || value.startsWith('8'))) {
      value = value.substring(1);
    }
    value = value.substring(0, 10);
    
    let formatted = '+7 (';
    if (value.length > 0) formatted += value.substring(0, 3);
    if (value.length > 3) formatted += ') ' + value.substring(3, 6);
    if (value.length > 6) formatted += '-' + value.substring(6, 8);
    if (value.length > 8) formatted += '-' + value.substring(8, 10);
    
    setFormData({ ...formData, phone: formatted });
  };

  if (isSuccess) {
    return (
      <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
        <div className="bg-white rounded-xl p-6 max-w-md w-full shadow-xl text-center">
          <h2 className="text-xl font-bold mb-4 text-green-600">Запись подтверждена!</h2>
          <p className="mb-4">Вы записаны на {slot.groupLevel} ({slot.groupTeacher}) на {slot.dateStr} в {slot.time}.</p>
          <button onClick={onSuccess} className="w-full py-2 bg-blue-600 text-white rounded-lg">Закрыть</button>
        </div>
      </div>
    );
  }

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
      <div className="bg-white rounded-xl p-6 max-w-md w-full shadow-xl">
        <h2 className="text-xl font-bold mb-4">Запись на пробный урок</h2>
        
        {existingSignup ? (
          <div className="mb-4 p-4 bg-yellow-50 border border-yellow-200 rounded-lg">
            <p className="text-sm text-yellow-800 mb-2">У вас уже есть запись:</p>
            <p className="font-bold text-sm">{existingSignup.groupLevel} на {existingSignup.date}</p>
            <p className="text-sm mt-2">Вы хотите перезаписаться на текущее занятие?</p>
            <div className="flex gap-2 mt-4">
              <button onClick={() => setExistingSignup(null)} className="flex-1 py-1 border rounded" disabled={isSubmitting}>Нет</button>
              <button onClick={() => {
                setIsSubmitting(true);
                saveSignup({ 
                  ...formData, 
                  groupId: slot.groupId, 
                  groupLevel: slot.groupLevel, 
                  groupTeacher: slot.groupTeacher, 
                  date: slotDateISO, 
                  time: slot.time 
                });
                setIsSuccess(true);
                setIsSubmitting(false);
              }} className="flex-1 py-1 bg-blue-600 text-white rounded" disabled={isSubmitting}>Да, перезаписаться</button>
            </div>
          </div>
        ) : (
          <form onSubmit={handleSubmit} className="space-y-4">
            <div>
              <label htmlFor="name" className="block text-sm font-medium text-gray-700">Имя</label>
              <input id="name" type="text" required className="w-full border rounded-lg p-2 mt-1" value={formData.name} onChange={(e) => setFormData({ ...formData, name: e.target.value })} disabled={isSubmitting} />
            </div>
            <div>
              <label htmlFor="phone" className="block text-sm font-medium text-gray-700">Телефон</label>
              <input id="phone" type="tel" required placeholder="+7 (XXX) XXX-XX-XX" className="w-full border rounded-lg p-2 mt-1" value={formData.phone} onChange={handlePhoneChange} disabled={isSubmitting} />
              {errors.phone && <p className="text-red-500 text-xs mt-1">{errors.phone}</p>}
            </div>
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">Email (опционально)</label>
              <input id="email" type="email" className="w-full border rounded-lg p-2 mt-1" value={formData.email} onChange={(e) => setFormData({ ...formData, email: e.target.value })} disabled={isSubmitting} />
            </div>
            <div className="flex gap-3 mt-6">
              <button type="button" onClick={onClose} className="flex-1 py-2 border rounded-lg hover:bg-gray-50" disabled={isSubmitting}>Отмена</button>
              <button type="submit" className="flex-1 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700" disabled={isSubmitting}>Записаться</button>
            </div>
          </form>
        )}
      </div>
    </div>
  );
}
````

````
const STORAGE_KEY = 'trial_signups';

export const getSignups = () => {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  } catch (e) {
    return [];
  }
};

export const saveSignup = (signup) => {
  const signups = getSignups();
  
  // Помечаем старую активную запись этого пользователя как отмененную
  const updatedSignups = signups.map(s => 
    s.phone === signup.phone && s.status === 'active' 
      ? { ...s, status: 'cancelled' } 
      : s
  );

  // Добавляем новую запись
  updatedSignups.push({ 
    ...signup, 
    id: Date.now().toString(), 
    status: 'active' 
  });
  
  localStorage.setItem(STORAGE_KEY, JSON.stringify(updatedSignups));
};
````

````
        <TrialSignupForm 
          slot={selectedSlot} 
          onClose={() => setSelectedSlot(null)}
          onSuccess={() => {
            setSignups(getSignups());
            setSelectedSlot(null);
          }}
        />
      )}
    </div>
  );
}
````
